cmake_minimum_required(VERSION 3.16)
project(capstone_fleet_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_package(capstone_fleet REQUIRED)
find_package(capstone_mission REQUIRED)
add_library(fleet_runtime_core src/scene.cpp src/robot_navigator.cpp src/fleet_jobs.cpp src/fleet_node.cpp)
target_include_directories(fleet_runtime_core PUBLIC include)
ament_target_dependencies(fleet_runtime_core rclcpp rclcpp_action nav_msgs nav2_msgs
  geometry_msgs diagnostic_msgs std_srvs)
target_link_libraries(fleet_runtime_core capstone_fleet::capstone_fleet_planning
  capstone_mission::capstone_mission_domain
  yaml-cpp Threads::Threads)
target_compile_options(fleet_runtime_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fleet_coordinator src/main.cpp)
target_link_libraries(fleet_coordinator fleet_runtime_core)
install(TARGETS fleet_coordinator DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scenarios DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_fleet_runtime test/test_runtime.cpp TIMEOUT 30)
  target_link_libraries(test_fleet_runtime fleet_runtime_core)
  target_compile_definitions(test_fleet_runtime PRIVATE
    FLEET_RUNTIME_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  set_tests_properties(test_fleet_runtime PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=177")
endif()
ament_package()
