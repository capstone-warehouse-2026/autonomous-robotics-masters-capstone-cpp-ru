cmake_minimum_required(VERSION 3.16)
project(capstone_webots_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)

# Обычная сборка G5 не требует SDK симулятора; физический профиль задаёт WEBOTS_HOME.
find_path(WEBOTS_INCLUDE_DIR webots/robot.h HINTS "$ENV{WEBOTS_HOME}/include/controller/c")
find_library(WEBOTS_CONTROLLER_LIBRARY Controller HINTS "$ENV{WEBOTS_HOME}/lib/controller")
if(WEBOTS_INCLUDE_DIR AND WEBOTS_CONTROLLER_LIBRARY)
  find_package(rclcpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(rosgraph_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(tf2_ros REQUIRED)
  add_executable(webots_bridge src/webots_bridge.cpp)
  target_include_directories(webots_bridge PRIVATE include ${WEBOTS_INCLUDE_DIR})
  target_link_libraries(webots_bridge ${WEBOTS_CONTROLLER_LIBRARY})
  ament_target_dependencies(webots_bridge rclcpp geometry_msgs nav_msgs sensor_msgs
    rosgraph_msgs std_msgs std_srvs tf2_ros)
  target_compile_options(webots_bridge PRIVATE -Wall -Wextra -Wpedantic)
  get_filename_component(WEBOTS_LIBRARY_DIR "${WEBOTS_CONTROLLER_LIBRARY}" DIRECTORY)
  set_target_properties(webots_bridge PROPERTIES INSTALL_RPATH "${WEBOTS_LIBRARY_DIR}")
  install(TARGETS webots_bridge DESTINATION lib/${PROJECT_NAME})
  add_executable(bridge_probe test/bridge_probe.cpp)
  ament_target_dependencies(bridge_probe rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs)
  target_compile_options(bridge_probe PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS bridge_probe DESTINATION lib/${PROJECT_NAME})
else()
  message(STATUS "Webots SDK absent: set WEBOTS_HOME and rebuild for physical simulation")
endif()
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../simulation/integration/"
  DESTINATION share/${PROJECT_NAME}/worlds)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../simulation/fleet/"
  DESTINATION share/${PROJECT_NAME}/fleet)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_wheel_odometry test/test_wheel_odometry.cpp)
  target_include_directories(test_wheel_odometry PRIVATE include)
endif()
ament_package()
