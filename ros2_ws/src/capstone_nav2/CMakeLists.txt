cmake_minimum_required(VERSION 3.16)
project(capstone_nav2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp geometry_msgs sensor_msgs nav_msgs diagnostic_msgs std_srvs rosgraph_msgs tf2 tf2_ros)
  find_package(${dep} REQUIRED)
endforeach()
add_library(scan_grid STATIC src/scan_grid.cpp)
target_include_directories(scan_grid PUBLIC include)
ament_target_dependencies(scan_grid nav_msgs sensor_msgs)
target_compile_options(scan_grid PRIVATE -Wall -Wextra -Wpedantic)
add_library(contracts_adapter STATIC src/contracts_adapter.cpp)
target_include_directories(contracts_adapter PUBLIC include)
target_link_libraries(contracts_adapter scan_grid)
ament_target_dependencies(contracts_adapter rclcpp geometry_msgs sensor_msgs nav_msgs
  diagnostic_msgs std_srvs rosgraph_msgs tf2 tf2_ros)
target_compile_options(contracts_adapter PRIVATE -Wall -Wextra -Wpedantic)
add_executable(nav2_contracts src/main.cpp)
target_link_libraries(nav2_contracts contracts_adapter)
install(TARGETS nav2_contracts DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_scan_grid test/test_scan_grid.cpp)
  target_link_libraries(test_scan_grid scan_grid)
  ament_add_gtest(test_contracts_adapter test/test_contracts_adapter.cpp)
  target_link_libraries(test_contracts_adapter contracts_adapter)
  set_tests_properties(test_contracts_adapter PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=167")
endif()
ament_package()
