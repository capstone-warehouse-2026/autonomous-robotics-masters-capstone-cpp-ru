cmake_minimum_required(VERSION 3.16)
project(capstone_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(capstone_mission REQUIRED)
foreach(dep std_srvs rosgraph_msgs nav_msgs diagnostic_msgs)
  find_package(${dep} REQUIRED)
endforeach()
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(${PROJECT_NAME}_core STATIC src/benchmark_runner_node.cpp)
target_include_directories(${PROJECT_NAME}_core PUBLIC include)
ament_target_dependencies(${PROJECT_NAME}_core rclcpp std_srvs rosgraph_msgs nav_msgs diagnostic_msgs)
target_link_libraries(${PROJECT_NAME}_core yaml-cpp)
add_executable(benchmark_runner_node src/main.cpp)
target_link_libraries(benchmark_runner_node ${PROJECT_NAME}_core)
add_library(dispatch_metrics STATIC src/episode.cpp)
target_include_directories(dispatch_metrics PUBLIC include)
ament_target_dependencies(dispatch_metrics capstone_mission)
target_compile_options(dispatch_metrics PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dispatch_benchmark src/dispatch_benchmark.cpp)
target_link_libraries(dispatch_benchmark dispatch_metrics yaml-cpp OpenSSL::Crypto)
add_executable(aggregate_dispatch src/aggregate.cpp)
target_link_libraries(aggregate_dispatch dispatch_metrics)
add_executable(mission_metrics src/mission_metrics.cpp)
target_link_libraries(mission_metrics dispatch_metrics yaml-cpp)
install(TARGETS benchmark_runner_node dispatch_benchmark dispatch_metrics aggregate_dispatch mission_metrics DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_benchmark_runner_node test/test_benchmark_runner_node.cpp)
  target_link_libraries(test_benchmark_runner_node ${PROJECT_NAME}_core)
  set_tests_properties(test_benchmark_runner_node PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=179")
  ament_add_gtest(test_episode test/test_episode.cpp)
  target_link_libraries(test_episode dispatch_metrics)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_mission_metrics_cli
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_mission_metrics.py
            $<TARGET_FILE:mission_metrics>)
endif()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
