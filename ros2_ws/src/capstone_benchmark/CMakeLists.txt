cmake_minimum_required(VERSION 3.16)
project(capstone_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)

# Node code lives in a library so that the executable and the tests share it.
add_library(${PROJECT_NAME}_core STATIC src/benchmark_runner_node.cpp)
set_target_properties(${PROJECT_NAME}_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(${PROJECT_NAME}_core PUBLIC include)
ament_target_dependencies(${PROJECT_NAME}_core rclcpp)
target_compile_options(${PROJECT_NAME}_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(benchmark_runner_node src/main.cpp)
target_link_libraries(benchmark_runner_node ${PROJECT_NAME}_core)
target_compile_options(benchmark_runner_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS benchmark_runner_node DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_benchmark_runner_node test/test_benchmark_runner_node.cpp)
  target_link_libraries(test_benchmark_runner_node ${PROJECT_NAME}_core)
endif()

# G1 episode evaluator: pure C++ metrics; no ROS graph or simulator needed.
add_library(g1_eval_core STATIC src/g1_eval.cpp)
target_include_directories(g1_eval_core PUBLIC include)
target_compile_options(g1_eval_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(g1_trajectory_eval src/g1_eval_main.cpp)
target_link_libraries(g1_trajectory_eval PRIVATE g1_eval_core)
target_compile_options(g1_trajectory_eval PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS g1_trajectory_eval DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  ament_add_gtest(test_g1_eval test/test_g1_eval.cpp)
  target_link_libraries(test_g1_eval g1_eval_core)
endif()

add_library(g1_batch_core STATIC src/g1_batch.cpp)
target_link_libraries(g1_batch_core PUBLIC g1_eval_core)
target_compile_options(g1_batch_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(g1_batch_eval src/g1_batch_main.cpp)
target_link_libraries(g1_batch_eval PRIVATE g1_batch_core)
target_compile_options(g1_batch_eval PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS g1_batch_eval DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  ament_add_gtest(test_g1_batch test/test_g1_batch.cpp)
  target_link_libraries(test_g1_batch g1_batch_core)
endif()

ament_package()
