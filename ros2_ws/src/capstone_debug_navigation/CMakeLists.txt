cmake_minimum_required(VERSION 3.16)
project(capstone_debug_navigation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action nav2_msgs nav_msgs geometry_msgs rosgraph_msgs diagnostic_msgs std_srvs)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(debug_navigation src/debug_navigation.cpp)
ament_target_dependencies(debug_navigation rclcpp rclcpp_action nav2_msgs nav_msgs geometry_msgs rosgraph_msgs diagnostic_msgs std_srvs)
target_compile_options(debug_navigation PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS debug_navigation DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
