cmake_minimum_required(VERSION 3.16)
project(capstone_system_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(system_evaluator src/system_evaluator.cpp)
target_include_directories(system_evaluator PRIVATE include)
ament_target_dependencies(system_evaluator rclcpp nav_msgs diagnostic_msgs std_srvs std_msgs)
target_link_libraries(system_evaluator yaml-cpp)
target_compile_options(system_evaluator PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS system_evaluator DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scenarios DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_route_evidence test/test_route_evidence.cpp)
  target_include_directories(test_route_evidence PRIVATE include)
endif()
ament_package()
